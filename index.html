        // Initialize Firebase

        async function initFirebaseSync() {

            try {

                if (typeof __firebase_config !== 'undefined') {

                    const firebaseConfig = JSON.parse(__firebase_config);

                    appId = typeof __app_id !== 'undefined' ? __app_id : 'skyhigh-obby-3d';

                    const app = initializeApp(firebaseConfig);

                    db = getFirestore(app);

                    auth = getAuth(app);


                    if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {

                        await signInWithCustomToken(auth, __initial_auth_token);

                    } else {

                        await signInAnonymously(auth);

                    }

                    userId = auth.currentUser?.uid || crypto.randomUUID();


                    const playersCol = collection(db, 'artifacts', appId, 'public', 'data', 'players');

                    onSnapshot(playersCol, (snapshot) => {

                        let count = 0;

                        snapshot.forEach((docSnap) => {

                            const data = docSnap.data();

                            if (docSnap.id !== userId) {

                                count++;

                                updateRemotePlayer(docSnap.id, data);

                            }

                        });

                        document.getElementById('player-count').innerText = `Online Players: ${count + 1}`;

                    }, (err) => console.log(err));

                }

            } catch (e) {

                console.log("Offline mode active:", e);

            }

        }


        function updateRemotePlayer(id, data) {

            if (!remotePlayers[id]) {

                remotePlayers[id] = createAvatarMesh(data.avatar || 'boy');

                scene.add(remotePlayers[id]);

            }

            remotePlayers[id].position.set(data.x, data.y, data.z);

            remotePlayers[id].rotation.y = data.ry || 0;

            if (data.swinging && remotePlayers[id].swingBat) {

                remotePlayers[id].swingBat();

            }

        }

        let scene, camera, renderer;

        let platformMeshes = [], obstacles = [], checkpoints = [];

        let lastCheckpointPos = new THREE.Vector3(0, 2, 0);

        let doubleJumpRings = [];

        

        // Dynamic Player Physics State

        let playerVelocity = new THREE.Vector3(0, 0, 0);

        let isGrounded = false;

        const MAX_JUMPS = Infinity;

        const BASE_JUMP_FORCE = 0.35;

        const PLAYER_RADIUS = 0.4;

        const PLAYER_HEIGHT = 2.0;

        let isSwinging = false, swingTimer = 0;


        function initThreeGame() {

            scene = new THREE.Scene();

            scene.background = new THREE.Color(0x0f172a);

            scene.fog = new THREE.FogExp2(0x0f172a, 0.012);


            camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 1000);

            

            renderer = new THREE.WebGLRenderer({ antialias: true });

            renderer.setSize(window.innerWidth, window.innerHeight);

            renderer.shadowMap.enabled = true;

            renderer.shadowMap.type = THREE.PCFSoftShadowMap;

            document.body.appendChild(renderer.domElement);


            // Lighting

            const ambientLight = new THREE.AmbientLight(0xffffff, 0.7);

            scene.add(ambientLight);


            const dirLight = new THREE.DirectionalLight(0xffffff, 0.9);

            dirLight.position.set(40, 80, 40);

            dirLight.castShadow = true;

            dirLight.shadow.mapSize.width = 2048;

            dirLight.shadow.mapSize.height = 2048;

            scene.add(dirLight);


            // Build 25 Obby Stages

            buildObbyStages();


            // Create Player Mesh

            playerMesh = createAvatarMesh(selectedAvatarType);

            playerMesh.position.copy(lastCheckpointPos);

            scene.add(playerMesh);


            // Controls listeners

            window.addEventListener('resize', onWindowResize);

            window.addEventListener('keydown', (e) => handleKey(e, true));

            window.addEventListener('keyup', (e) => handleKey(e, false));

            

            window.addEventListener('mousedown', (e) => {

                if (e.button === 0 && e.target.tagName !== 'INPUT' && e.target.tagName !== 'BUTTON') {

                    mouse.isDragging = true;

                    mouse.prevX = e.clientX;

                    mouse.prevY = e.clientY;

                    triggerBatSwing();

                }

            });

            window.addEventListener('mousemove', (e) => {

                if (mouse.isDragging) {

                    const deltaX = e.clientX - mouse.prevX;

                    const deltaY = e.clientY - mouse.prevY;

                    mouse.theta -= deltaX * 0.005;

                    mouse.phi = Math.max(0.05, Math.min(1.3, mouse.phi - deltaY * 0.005));

                    mouse.prevX = e.clientX;

                    mouse.prevY = e.clientY;

                }

            });

            window.addEventListener('mouseup', () => mouse.isDragging = false);


            animateGame();

        }


        function createAvatarMesh(type, isGold = false) {

            const group = new THREE.Group();

            const isBoy = type === 'boy';


            // Body

            const bodyMat = new THREE.MeshStandardMaterial({ color: isBoy ? 0x3b82f6 : 0xec4899, roughness: 0.3 });

            const body = new THREE.Mesh(new THREE.BoxGeometry(0.8, 1.0, 0.5), bodyMat);

            body.position.y = 1.0;

            body.castShadow = true;

            group.add(body);


            // Head

            const headMat = new THREE.MeshStandardMaterial({ color: 0xfde047, roughness: 0.4 });

            const head = new THREE.Mesh(new THREE.BoxGeometry(0.7, 0.7, 0.7), headMat);

            head.position.y = 1.85;

            head.castShadow = true;

            group.add(head);


            // Hair / Cap

            const hairMat = new THREE.MeshStandardMaterial({ color: isBoy ? 0x1e293b : 0x831843 });

            const hair = new THREE.Mesh(new THREE.BoxGeometry(0.75, 0.25, 0.75), hairMat);

            hair.position.y = 2.25;

            group.add(hair);


            // Bat Weapon in Hand

            const batGroup = new THREE.Group();

            const batMat = new THREE.MeshStandardMaterial({

                color: isGold ? 0xf59e0b : 0xd97706,

                metalness: isGold ? 0.8 : 0.1,

                roughness: isGold ? 0.2 : 0.4

            });

            const bat = new THREE.Mesh(new THREE.CylinderGeometry(0.08, 0.04, 1.2, 8), batMat);

            bat.rotation.z = -Math.PI / 4;

            bat.position.set(0.6, 1.0, 0.3);

            bat.castShadow = true;

            batGroup.add(bat);

            group.add(batGroup);


            group.batGroup = batGroup;

            group.swingBat = () => {

                isSwinging = true;

                swingTimer = 12;

            };


            return group;

        }


        // --- 25 STAGES BUILDER ---

        function buildObbyStages() {

            platformMeshes = [];

            obstacles = [];

            checkpoints = [];


            let currentZ = 0;

            const stageColors = [

                0x3b82f6, 0x10b981, 0x8b5cf6, 0xf59e0b, 0xec4899,

                0x06b6d4, 0x84cc16, 0x6366f1, 0xd97706, 0xf43f5e

            ];


            // Helper to add platforms

            function addPlatform(x, y, z, w, h, d, color = 0x3b82f6, isMoving = false, amp = 3, bounce = false, speed = false) {

                const geom = new THREE.BoxGeometry(w, h, d);

                let matColor = color;

                if (bounce) matColor = 0xa855f7;

                if (speed) matColor = 0x06b6d4;


                const mat = new THREE.MeshStandardMaterial({ color: matColor, roughness: 0.3 });

                const mesh = new THREE.Mesh(geom, mat);

                mesh.position.set(x, y, z);

                mesh.receiveShadow = true;

                mesh.castShadow = true;


                mesh.userData = {

                    isMoving: isMoving,

                    initialX: x,

                    amplitude: amp,

                    isBouncePad: bounce,

                    isSpeedPad: speed

                };


                scene.add(mesh);

                platformMeshes.push(mesh);

                return mesh;

            }


            // Helper to add checkpoints

            function addCheckpoint(stageNum, x, y, z) {

                const pad = addPlatform(x, y, z, 8, 0.8, 8, 0x22c55e);

                

                // Flag pole

                const poleGeom = new THREE.CylinderGeometry(0.1, 0.1, 4, 8);

                const poleMat = new THREE.MeshStandardMaterial({ color: 0xe2e8f0 });

                const pole = new THREE.Mesh(poleGeom, poleMat);

                pole.position.set(x - 3, y + 2, z - 3);

                scene.add(pole);


                const flagGeom = new THREE.BoxGeometry(1.5, 1, 0.1);

                const flagMat = new THREE.MeshStandardMaterial({ color: 0xef4444 });

                const flag = new THREE.Mesh(flagGeom, flagMat);

                flag.position.set(x - 2.2, y + 3.5, z - 3);

                scene.add(flag);


                checkpoints.push({ stage: stageNum, pos: new THREE.Vector3(x, y + 1.5, z) });

            }


            // Helper to add lava obstacles

            function addLava(x, y, z, w, h, d) {

                const geom = new THREE.BoxGeometry(w, h, d);

                const mat = new THREE.MeshBasicMaterial({ color: 0xef4444 });

                const mesh = new THREE.Mesh(geom, mat);

                mesh.position.set(x, y, z);

                mesh.userData = { isLava: true, rotSpeed: 0.02 };

                scene.add(mesh);

                obstacles.push(mesh);

            }


            // Spawn Platform (Stage 1 Start)

            addCheckpoint(1, 0, 0, currentZ);


            // Build 25 Stages procedurally with distinct challenges

            for (let s = 1; s <= 25; s++) {

                const color = stageColors[s % stageColors.length];

                

                // Add Checkpoint at start of stage (every stage or step)

                if (s > 1) {

                    addCheckpoint(s, 0, 0, currentZ);

                }


                // Stage platform patterns based on stage number

                const numSteps = 5;

                for (let step = 0; step < numSteps; step++) {

                    currentZ -= 12;

                    const offsetX = (step % 2 === 0 ? 1 : -1) * (s % 3 === 0 ? 3 : 0);


                    if (s % 5 === 0) {

                        // Moving Platforms Stage

                        addPlatform(offsetX, 0, currentZ, 4, 0.6, 4, color, true, 4 + (s % 3));

                    } else if (s % 4 === 0) {

                        // Bounce Pad & High Jumps

                        addPlatform(offsetX, 0, currentZ, 3.5, 0.6, 3.5, color, false, 0, step === 2);

                        if (step === 2) addLava(offsetX, 1.2, currentZ, 1.5, 0.6, 1.5);

                    } else if (s % 3 === 0) {

                        // Speed Boost & Narrow Balance Beam

                        addPlatform(offsetX, 0, currentZ, 2.5, 0.6, 6, color, false, 0, false, step % 2 === 0);

                        if (step % 2 === 1) addLava(offsetX, 0.8, currentZ, 2.5, 0.4, 0.8);

                    } else {

                        // Standard Jump Stepping Stones

                        addPlatform(offsetX, (step * 0.5), currentZ, 4 - (step * 0.3), 0.6, 4, color);

                    }

                }


                currentZ -= 15;

            }


            // Final Victory Podium (Stage 25 completion)

            addPlatform(0, 0, currentZ, 16, 1, 16, 0xeab308);

            const trophyGeom = new THREE.TorusGeometry(2, 0.5, 16, 32);

            const trophyMat = new THREE.MeshStandardMaterial({ color: 0xf59e0b, metalness: 0.9, roughness: 0.1 });

            const trophy = new THREE.Mesh(trophyGeom, trophyMat);

            trophy.position.set(0, 4, currentZ);

            scene.add(trophy);


            window.victoryZ = currentZ + 5;

        }


        function spawnDoubleJumpEffect(pos) {

            const ringGeom = new THREE.RingGeometry(0.2, 1.1, 16);

            const ringMat = new THREE.MeshBasicMaterial({ color: 0x38bdf8, side: THREE.DoubleSide, transparent: true, opacity: 0.9 });

            const ring = new THREE.Mesh(ringGeom, ringMat);

            ring.rotation.x = Math.PI / 2;

            ring.position.copy(pos);

            ring.position.y += 0.1;

            scene.add(ring);

            doubleJumpRings.push({ mesh: ring, scale: 1.0, opacity: 0.9 });

        }


        // --- RIGID PHYSICS & COLLISION SOLVER ---

        function handleKey(e, isDown) {

            if (e.code === 'KeyW' || e.code === 'ArrowUp') keys.w = isDown;

            if (e.code === 'KeyA' || e.code === 'ArrowLeft') keys.a = isDown;

            if (e.code === 'KeyS' || e.code === 'ArrowDown') keys.s = isDown;

            if (e.code === 'KeyD' || e.code === 'ArrowRight') keys.d = isDown;

            if (e.code === 'KeyQ') keys.q = isDown;

            if (e.code === 'KeyE') keys.e = isDown;

            

            if (e.code === 'Space') {

                if (isDown && !keys.space) {

                    if (isFlyMode) {

                        playerVelocity.y = 0.3;

                    } else if (isGrounded || isUnlimitedJump || jumpsCount < 2) {

                        const currentJumpForce = BASE_JUMP_FORCE * (isSuperJump ? 1.7 : 1.0);

                        playerVelocity.y = currentJumpForce;

                        jumpsCount++;

                        if (jumpsCount >= 2) {

                            spawnDoubleJumpEffect(playerMesh.position);

                        }

                        isGrounded = false;

                    }

                }

                keys.space = isDown;

            }

        }


        window.triggerBatSwing = () => {

            if (playerMesh && playerMesh.swingBat) {

                playerMesh.swingBat();

                checkBatHits();

            }

        };


        window.respawnPlayer = () => {

            if (isGodMode) return;

            playerMesh.position.copy(lastCheckpointPos);

            playerVelocity.set(0, 0, 0);

            jumpsCount = 0;

            deaths++;

            document.getElementById('hud-deaths').innerText = deaths;

        };


        function checkBatHits() {

            const knockbackPower = hasGoldenBat ? 12 : 5;

            for (let id in remotePlayers) {

                const rp = remotePlayers[id];

                if (playerMesh.position.distanceTo(rp.position) < 3.2) {

                    rp.position.z -= knockbackPower;

                    rp.position.y += 2.5;

                }

            }

        }

        function animateGame() {

            requestAnimationFrame(animateGame);


            if (!gameStarted) return;


            const time = Date.now() * 0.003;


            // Animate Double Jump Rings

            for (let i = doubleJumpRings.length - 1; i >= 0; i--) {

                const r = doubleJumpRings[i];

                r.scale += 0.06;

                r.opacity -= 0.05;

                r.mesh.scale.set(r.scale, r.scale, 1);

                r.mesh.material.opacity = r.opacity;

                if (r.opacity <= 0) {

                    scene.remove(r.mesh);

                    doubleJumpRings.splice(i, 1);

                }

            }


            // Animate Platforms & Obstacles

            platformMeshes.forEach(p => {

                if (p.userData && p.userData.isMoving) {

                    p.position.x = p.userData.initialX + Math.sin(time) * p.userData.amplitude;

                }

            });


            obstacles.forEach(obs => {

                if (obs.userData && obs.userData.isLava) {

                    obs.rotation.y += obs.userData.rotSpeed;


                    // Lava hit detection

                    if (!isGodMode) {

                        const obsBox = new THREE.Box3().setFromObject(obs);

                        const pBox = new THREE.Box3().setFromObject(playerMesh);

                        if (obsBox.intersectsBox(pBox)) {

                            respawnPlayer();

                        }

                    }

                }

            });


            // Bat Animation logic

            if (isSwinging && playerMesh.batGroup) {

                playerMesh.batGroup.rotation.x += 0.35;

                swingTimer--;

                if (swingTimer <= 0) {

                    isSwinging = false;

                    playerMesh.batGroup.rotation.x = 0;

                }

            }


            // Directional Movement relative to camera perspective

            let inputX = 0, inputZ = 0;

            let moveSpeed = 0.16 * speedMultiplier;


            if (keys.w) inputZ -= 1;

            if (keys.s) inputZ += 1;

            if (keys.a) inputX -= 1;

            if (keys.d) inputX += 1;


            if (inputX !== 0 || inputZ !== 0) {

                const moveAngle = Math.atan2(inputX, inputZ) + mouse.theta;

                playerVelocity.x = Math.sin(moveAngle) * moveSpeed;

                playerVelocity.z = Math.cos(moveAngle) * moveSpeed;

                playerMesh.rotation.y = moveAngle;

            } else {

                playerVelocity.x *= 0.7; // friction

                playerVelocity.z *= 0.7;

            }


            if (isFlyMode) {

                // Flying controls (Space = up, Q = down)

                if (keys.space) playerVelocity.y = 0.25 * speedMultiplier;

                else if (keys.q) playerVelocity.y = -0.25 * speedMultiplier;

                else playerVelocity.y *= 0.8;

                isGrounded = false;

                jumpsCount = 0;

            } else {

                // Apply Gravity (Strictly pull down every frame)

                playerVelocity.y -= GRAVITY;

            }


            // Apply horizontal velocity

            playerMesh.position.x += playerVelocity.x;

            playerMesh.position.z += playerVelocity.z;

            

            // Apply vertical velocity

            playerMesh.position.y += playerVelocity.y;


            // Ground & Platform Collision Resolution

            if (!isFlyMode) {

                isGrounded = false;

                const feetY = playerMesh.position.y;


                for (let i = 0; i < platformMeshes.length; i++) {

                    const platform = platformMeshes[i];

                    const box = new THREE.Box3().setFromObject(platform);

                    

                    // Horizontal bounds check

                    if (playerMesh.position.x >= box.min.x - PLAYER_RADIUS &&

                        playerMesh.position.x <= box.max.x + PLAYER_RADIUS &&

                        playerMesh.position.z >= box.min.z - PLAYER_RADIUS &&

                        playerMesh.position.z <= box.max.z + PLAYER_RADIUS) {

                        

                        // Landing on top of platform surface

                        if (feetY >= box.max.y - 0.3 && feetY <= box.max.y + 0.4 && playerVelocity.y <= 0) {

                            playerMesh.position.y = box.max.y;

                            playerVelocity.y = 0;

                            isGrounded = true;

                            jumpsCount = 0; // Reset double jump upon landing


                            // Special platform mechanics

                            if (platform.userData.isBouncePad) {

                                playerVelocity.y = BASE_JUMP_FORCE * 1.6; // High bounce

                                isGrounded = false;

                            } else if (platform.userData.isSpeedPad) {

                                playerVelocity.z -= 0.3; // Speed boost launch forward

                            }

                            break;

                        }

                    }

                }

            }


            // Checkpoints Check

            checkpoints.forEach(cp => {

                if (playerMesh.position.distanceTo(cp.pos) < 2.5) {

                    if (currentStage < cp.stage) {

                        currentStage = cp.stage;

                        lastCheckpointPos.copy(cp.pos);

                        document.getElementById('hud-stage').innerText = `${currentStage} / 25`;

                    }

                }

            });


            // Fall Void Bounds Reset Trigger

            if (playerMesh.position.y < -15 && !isGodMode && !isFlyMode) {

                respawnPlayer();

            }


            // Victory Check

            if (playerMesh.position.z <= (window.victoryZ || -600)) {

                gameStarted = false;

                document.getElementById('hud').classList.add('hidden');

                document.getElementById('victory-modal').classList.remove('hidden');

                document.getElementById('victory-time').innerText = document.getElementById('hud-timer').innerText;

                document.getElementById('victory-deaths').innerText = deaths;

            }


            // Live Timer Update

            const elapsed = Math.floor((Date.now() - startTime) / 1000);

            const mins = String(Math.floor(elapsed / 60)).padStart(2, '0');

            const secs = String(elapsed % 60).padStart(2, '0');

            const ms = String(Math.floor((Date.now() - startTime) % 1000 / 100));

            document.getElementById('hud-timer').innerText = `${mins}:${secs}.${ms}`;


            // Smooth Camera Follow

            const targetCamX = playerMesh.position.x + Math.sin(mouse.theta) * mouse.distance * Math.cos(mouse.phi);

            const targetCamY = playerMesh.position.y + Math.sin(mouse.phi) * mouse.distance + 1.8;

            const targetCamZ = playerMesh.position.z + Math.cos(mouse.theta) * mouse.distance * Math.cos(mouse.phi);


            camera.position.lerp(new THREE.Vector3(targetCamX, targetCamY, targetCamZ), 0.2);

            camera.lookAt(playerMesh.position.clone().add(new THREE.Vector3(0, 1.2, 0)));


            // Firebase Network Sync

            if (db && userId) {

                setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'players', userId), {

                    name: playerName,

                    avatar: selectedAvatarType,

                    isAdmin: isAdmin,

                    x: playerMesh.position.x,

                    y: playerMesh.position.y,

                    z: playerMesh.position.z,

                    ry: playerMesh.rotation.y,

                    swinging: isSwinging,

                    updatedAt: Date.now()

                }).catch(() => {});

            }


            renderer.render(scene, camera);

        }


        function onWindowResize() {

            camera.aspect = window.innerWidth / window.innerHeight;

            camera.updateProjectionMatrix();

            renderer.setSize(window.innerWidth, window.innerHeight);

        }

        function appendChatMessage(sender, text, colorClass = "text-blue-400") {

            const chatContainer = document.getElementById('chat-messages');

            const div = document.createElement('div');

            div.innerHTML = `<span class="font-bold ${colorClass}">${sender}:</span> ${text}`;

            chatContainer.appendChild(div);

            chatContainer.scrollTop = chatContainer.scrollHeight;

        }


        window.sendChatMessage = () => {

            const input = document.getElementById('chat-input');

            const rawMsg = input.value.trim();

            if (rawMsg) {

                const lowerMsg = rawMsg.toLowerCase();

                if (lowerMsg === '/admin panel' || lowerMsg === '/admin') {

                    isAdmin = true;

                    document.getElementById('admin-hud-btn').classList.remove('hidden');

                    toggleAdminPanel(true);

                    appendChatMessage("👑 Admin System", "Admin privileges activated!", "text-amber-400");

                } else if (lowerMsg === '/help') {

                    appendChatMessage("System", "Commands: /admin panel (open admin menu), WASD to move, Space x2 to double jump.", "text-slate-300");

                } else {

                    const tag = isAdmin ? '<span class="text-xs px-1 bg-amber-500 text-slate-950 font-black rounded mr-1">ADMIN</span>' : '';

                    appendChatMessage(tag + playerName, escapeHtml(rawMsg));

                }

                input.value = '';

            }

        };


        function escapeHtml(text) {

            return text.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");

        }
